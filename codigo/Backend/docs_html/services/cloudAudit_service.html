 19    """Orquesta las auditorías (estática y con IA): recupera el scan, llama al AuditController, calcula los
 20    contadores por severidad y guarda el resultado como un nuevo AuditResult."""
 21    def __init__(self):
 22        self.scan_repository = ScanRepository()
 23        self.audit_repository = AuditRepository()

 26    async def static_audit_cloud_resources(self, scan_id: str, user_id: str):
 27        """Ejecuta el análisis estático sobre los recursos de un scan ya completado y guarda el resultado.
 28
 29        Args:
 30            scan_id (str): id del scan cuyos recursos se van a auditar.
 31            user_id (str): usuario que pide la auditoría, queda asociado al AuditResult creado.
 32
 33        Returns:
 34            AuditResult: resultado nuevo, con id propio, origin="static" y los counts por severidad ya calculados.
 35
 36        Raises:
 37            HTTPException: 404 si no existe un scan con ese id.
 38        """
 39        scanResult= await self.scan_repository.findById(scan_id)
 40        if not scanResult:
 41            logger.error(f"Scan with id {scan_id} not found for user {user_id}")
 42            raise HTTPException(status_code=404, detail="Scan not found")
 43
 44        resources = scanResult.resources
 45        accountId=scanResult.cloudAccount_id
 46
 47        deserializedResources=JSONDeserializer.deserialize_resources(resources)
 48        vulnerabilities=self.auditController.staticAudit(deserializedResources,"AWS")
 49        auditID=str(uuid4())
 50        vulnerabilities_serialized=JSONSerializer.serializeList(vulnerabilities)
 51        critical_count=sum(1 for v in vulnerabilities if v.severity.lower() == 'critical')
 52        high_count=sum(1 for v in vulnerabilities if v.severity.lower() == 'high')
 53        medium_count=sum(1 for v in vulnerabilities if v.severity.lower() == 'medium')
 54        low_count=sum(1 for v in vulnerabilities if v.severity.lower() == 'low')
 55        counts = {'critical': critical_count, 'high': high_count, 'medium': medium_count, 'low': low_count}
 56        auditResult= AuditResult(id=auditID, vulnerabilities=vulnerabilities_serialized, accountID=accountId,userID=user_id,resources=resources, origin=AuditOrigin.Static.value, counts=counts)
 57        insterdId=await self.audit_repository.create(auditResult)
 58        logger.info(f"Created audit result with id {insterdId} for scan {scan_id} and user {user_id}")
 59        return auditResult

 90            HTTPException: 404 si no existe el scan o la auditoría estática indicada.

 97            raise HTTPException(status_code=404, detail="Scan not found")

 99            logger.error(f"Audit result with id {audit_id} not found for user {user_id}")

110
111        auditAiResult= AuditResult(id=auditID, vulnerabilities=JSONSerializer.serializeList(response_vulnerabilities), accountID=scanResult.cloudAccount_id,userID=user_id,resources=resources,origin=AuditOrigin.AI.value, counts=counts)
112        insertedId=await self.audit_repository.create(auditAiResult)
113        return auditAiResult

145    async def generate_ai_recommendation(self, vulnerability_id:str, audit_id:str, user_id:str):
146        """Pide a Gemini la recomendación y el comando CLI para una vulnerabilidad concreta de una auditoría
147        ya guardada, y persiste el resultado en esa misma vulnerabilidad (para no tener que regenerarlo otra vez).
148
149        Args:
150            vulnerability_id (str): id de la vulnerabilidad dentro de la auditoría.
151            audit_id (str): id de la auditoría que contiene esa vulnerabilidad.
152            user_id (str): usuario que pide la recomendación, solo para logging.
153
154        Returns:
155            dict: {"recommendation": str, "cli_command": str} — lo mismo que devuelve GeminiAnalyzer.generateCLI.
156
157        Raises:
158            HTTPException: 404 si la vulnerabilidad no existe dentro de esa auditoría.
159        """
160        vulnerability= await self.audit_repository.findVulnerabilityById(audit_id, vulnerability_id)
161        if not vulnerability:
162            logger.error(f"Vulnerability with id {vulnerability_id} not found for user {user_id}")
163            raise HTTPException(status_code=404, detail="Vulnerability not found")
164        geminiAnalyzer=GeminiAnalyzer()
165        ai_recommendation=geminiAnalyzer.generateCLI(vulnerability)
166        vulnerability.recommendation=ai_recommendation.get("recommendation", "")
167        vulnerability.cli_command=ai_recommendation.get("cli_command", "")
168        await self.audit_repository.updateVulnerability(audit_id, vulnerability_id, vulnerability)
169        logger.info(f"Generated AI recommendation for vulnerability {vulnerability_id} in audit {audit_id} for user {user_id}")
170        return ai_recommendation

 23  def analyze(self, resources,vulnerabilities,userContext=dict):
 24    """Construye el prompt de auditoría, lo manda a Gemini y devuelve los hallazgos ya convertidos a Vulnerability.
 25
 26    Args:
 27        resources: recursos escaneados de la cuenta (dict crudo, no el modelo de dominio).
 28        vulnerabilities: hallazgos del análisis estático, se incluyen en el prompt para que el modelo no los repita.
 29        userContext (dict): contexto de negocio aportado por el usuario (campos "company" y "resources").
 30
 31    Returns:
 32        list[Vulnerability]: hallazgos adicionales que propone el modelo. Lista vacía si no encuentra nada nuevo.
 33
 34    Example:
 35        >>> GeminiAnalyzer().analyze(scan_resources, static_vulns, {"company": "fintech con datos de clientes"})
 36        [Vulnerability(id="ai_001", severity="High", ...)]
 37    """
 38    try:
 39      prompt=self.build_analyze_prompt(resources, vulnerabilities, userContext)
 40      response = self.client.models.generate_content(
 41        model="gemini-3.1-flash-lite-preview",
 42        contents=prompt
 43        )
 44      return self.parse_analyze_response(response.text)
 45    except Exception as e:
 46      logger.error(f"Error during Gemini analysis: {str(e)}")
 47      raise Exception(f"Error analyzing with Gemini : {str(e)}")

132  def parse_analyze_response(self, response):
133    """Limpia la respuesta de Gemini (a veces viene envuelta en \\`\\`\\`json ... \\`\\`\\`) y la convierte en
134    una lista de Vulnerability.
135
136    Args:
137        response (str): texto crudo devuelto por response.text.
138
139    Returns:
140        list[Vulnerability]: vacía si el modelo respondió "[]".
141
142    Raises:
143        Exception: si el texto no es JSON válido tras quitar el markdown — normalmente indica que el modelo
144            no siguió el formato pedido.
145    """
146    try:
147      text_received=response.strip()
148      if text_received.startswith("```"):
149          text_received=text_received.split("```")[1]
150          if text_received.startswith("json"):
151            text_received=text_received[4:]
152
153
154
155      lista=json.loads(text_received)
156      vulnerabilities=[]
157
158      for item in lista:
159        vulnerabilities.append(
160          Vulnerability(
161            id=item.get("id"),
162            name=item.get("name"),
163            description=item.get("description"),
164            severity=item.get("severity"),
165            resource_id=item.get("resource_id"),
166            resource_type=item.get("resource_type"),
167          )
168        )
169      return vulnerabilities
170    except Exception as e :
171      logger.error("error during parsing the response")
172      raise Exception(f"Error during parsing the response : {str(e)}")

191Responde SOLO en JSON:

198  def parse_generateCLI_response(self, response):
199    """Igual que parse_analyze_response pero para la respuesta de generateCLI: quita el bloque de código si lo
200    hay y valida que venga tanto el comando como la recomendación.
201
202    Args:
203        response (str): texto crudo de Gemini.
204
205    Returns:
206        dict | None: {"cli_command": str, "recommendation": str}. None si el modelo dice que no hay comando
207            (cli_command es null/vacío) o si falta la recomendación.
208    """
209    try:
210      text_received = response.strip()
211      if text_received.startswith("```"):
212        text_received = text_received.split("```")[1]
213        # strip language tag (json, bash, etc.)
214        first_newline = text_received.find("\n")
215        if first_newline != -1:
216          text_received = text_received[first_newline:].strip()
217
218      parsed = json.loads(text_received)
219      cmd = parsed.get("cli_command")
220      recommendation = parsed.get("recommendation")
221      if not cmd or str(cmd).lower() == "null":
222        return None
223      if not recommendation:
224        logger.warning("CLI command provided without recommendation")
225        return None
226      return {
227        "cli_command": cmd,
228        "recommendation": recommendation
229      }
230    except Exception as e:
231      logger.error("error during parsing the CLI response")
232      raise Exception(f"Error during parsing the CLI response : {str(e)}")

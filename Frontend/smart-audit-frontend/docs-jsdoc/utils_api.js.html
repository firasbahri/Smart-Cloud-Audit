/**
 * @module utils/api
 * @description Utilidades para construir URLs de la API REST y del endpoint SSE.
 * Lee la base de la URL desde la variable de entorno VITE_API_BASE_URL
 * (definida en .env / .env.production). Si no está definida, usa http://localhost:8000.
 */

const API_BASE_URL = (import.meta.env.VITE_API_BASE_URL || 'http://localhost:8000').replace(/\/+$/, '')

/**
 * Construye la URL completa para una llamada REST al backend.
 * Garantiza que haya exactamente una barra entre la base y el path,
 * independientemente de cómo venga el path.
 *
 * @param {string} [path=''] - Ruta del endpoint, con o sin barra inicial (ej. '/auth/login' o 'auth/login').
 * @returns {string} URL completa lista para usar en fetch() (ej. 'http://localhost:8000/auth/login').
 *
 * @example
 * buildApiUrl('/cloud/my-audits/123')
 * // → 'http://localhost:8000/cloud/my-audits/123'
 */
export const buildApiUrl = (path = '') => {
  const normalizedPath = String(path).startsWith('/') ? path : `/${path}`
  return `${API_BASE_URL}${normalizedPath}`
}

/**
 * Construye la URL para el endpoint SSE, añadiendo queryParams al query string.
 * Se usa principalmente para el endpoint de progreso de escaneo,
 * que recibe el token en la query porque EventSource no admite headers personalizados.
 *
 * @param {string} path - Ruta del endpoint SSE (ej. '/cloud/scan_progress_sse/{scanId}').
 * @param {Object} [queryParams={}] - Pares clave/valor a añadir como query string.
 *   Los valores undefined, null o cadena vacía se omiten.
 * @returns {string} URL completa con query string (ej. '…/scan_progress_sse/abc?token=xyz').
 *
 * @example
 * buildApiSseUrl('/cloud/scan_progress_sse/abc123', { token: 'eyJhbG...' })
 * // → 'http://localhost:8000/cloud/scan_progress_sse/abc123?token=eyJhbG...'
 */
export const buildApiSseUrl = (path, queryParams = {}) => {
  const url = new URL(buildApiUrl(path))

  Object.entries(queryParams).forEach(([key, value]) => {
    if (value !== undefined && value !== null && value !== '') {
      url.searchParams.set(key, String(value))
    }
  })

  return url.toString()
}

/**
 * @module scanStore
 * @description Store de Pinia para el ciclo de vida de los escaneos AWS.
 * Gestiona el progreso SSE de cada escaneo en curso, los resultados ya cargados,
 * el contexto IA por recurso y las notificaciones flotantes de ScanProgressToast.
 */

import { defineStore } from 'pinia'
import { ref } from 'vue'
import { buildApiSseUrl, buildApiUrl } from '../utils/api'

export const useScanStore = defineStore('scan', () => {

  /** @type {Object} Map accountId → scanId del último scan. */
  const scanResultIdByAccount = ref({})

  /** @type {Object} Map accountId → recursos del último scan. */
  const scanResultByAccount = ref({})

  /** @type {Object} Map accountId → progreso actual (0-100). */
  const scanProgressByAccount = ref({})

  /** @type {Object} Map accountId → true si hay escaneo en curso. */
  const scanningAccounts = ref({})

  /** @type {Object} Map accountId → scanId activo (guarda para evitar refetch). */
  const scanIdByAccount = ref({})

  /** @type {Object} Map accountId → ISO fecha de creación del último scan. */
  const scanCreatedAtByAccount = ref({})

  /** @type {Object} Cola de notificaciones pendientes de consumir por ScanProgressToast. */
  const pendingNotifications = ref([])

  /** @type {Object} Map accountId → map resourceId → texto de contexto IA. */
  const resourceContextsByAccount = ref({})

  // Map no reactivo: gestión interna de conexiones EventSource (no necesita reactividad)
  const eventSources = new Map()

  /**
   * Actualiza los datos de un scan completado o parcial para una cuenta.
   * @param {string} accountId
   * @param {string} scanId
   * @param {Object} dataResult - Recursos escaneados (users, groups, roles, ec2, buckets).
   * @param {string} createdAt - Fecha ISO del scan.
   * @param {Object} [context] - Contexto IA guardado en el scan (resourceId → texto).
   */
  const setScanData = (accountId, scanId, dataResult, createdAt, context) => {
    scanResultIdByAccount.value[accountId] = scanId
    scanResultByAccount.value[accountId] = dataResult
    scanCreatedAtByAccount.value[accountId] = createdAt
    resourceContextsByAccount.value[accountId] = context
    console.log('Datos de escaneo actualizados para la cuenta:', accountId, { scanId, createdAt, result: dataResult })
  }

  /**
   * Marca una cuenta como "escaneando" y registra el scanId activo.
   * Lo llama startSSE antes de abrir la conexión EventSource.
   * @param {string} accountId
   * @param {string} scanId
   */
  const startAccountScan = (accountId, scanId) => {
    scanningAccounts.value[accountId] = true
    scanProgressByAccount.value[accountId] = 0
    scanIdByAccount.value[accountId] = scanId
  }

  /**
   * Actualiza el progreso de un escaneo en curso, normalizado al rango [0, 100].
   * @param {string} accountId
   * @param {number} progress - Valor entre 0 y 100.
   */
  const setAccountScanProgress = (accountId, progress) => {
    const normalized = Number.isFinite(progress)
      ? Math.max(0, Math.min(100, Number(progress)))
      : 0
    scanProgressByAccount.value[accountId] = normalized
  }

  /**
   * Marca el escaneo de una cuenta como completado (progress=100, scanning=false).
   * @param {string} accountId
   */
  const completeAccountScan = (accountId) => {
    scanProgressByAccount.value[accountId] = 100
    scanningAccounts.value[accountId] = false
  }

  /**
   * Marca el escaneo de una cuenta como fallido sin tocar el progreso.
   * @param {string} accountId
   */
  const failAccountScan = (accountId) => {
    scanningAccounts.value[accountId] = false
  }

  /**
   * Guarda o elimina el contexto IA de un recurso concreto, y lo persiste en el backend
   * mediante PUT /cloud/update_context/{scan_id}.
   * @param {string} accountId
   * @param {string} resourceId - id del recurso dentro del inventario.
   * @param {string} context - Texto de contexto (vacío = eliminar el contexto existente).
   */
  const setResourceContext = (accountId, resourceId, context) => {
    const token = localStorage.getItem('token')
    if (!accountId) return
    if (!resourceContextsByAccount.value[accountId]) resourceContextsByAccount.value[accountId] = {}
    const trimmed = context.trim()
    if (trimmed) {
      resourceContextsByAccount.value[accountId][resourceId] = trimmed
    } else {
      delete resourceContextsByAccount.value[accountId][resourceId]
    }
    const scan_id = scanResultIdByAccount.value[accountId]
    console.log('Actualizando contexto para el recurso:', { accountId, resourceId, context: trimmed, scan_id })
    if (scan_id) {
      fetch(buildApiUrl(`/cloud/update_context/${scan_id}`), {
        method: 'PUT',
        headers: { 'Authorization': `Bearer ${token}`, 'Content-Type': 'application/json' },
        body: JSON.stringify({ resource_id: resourceId, context: context })
      })
    }
  }

  /**
   * Vacía por completo todos los datos de escaneo del store.
   * Se usa al cerrar sesión o al cambiar de cuenta seleccionada.
   */
  const clearData = () => {
    scanCreatedAtByAccount.value = {}
    scanResultIdByAccount.value = {}
    scanResultByAccount.value = {}
    scanProgressByAccount.value = {}
    scanningAccounts.value = {}
    scanIdByAccount.value = {}
  }

  /**
   * Encola una notificación para que ScanProgressToast la muestre.
   * @param {{severity: string, summary: string, detail: string, life: number}} notification
   */
  const pushNotification = (notification) => {
    pendingNotifications.value.push({ ...notification, _id: Date.now() + Math.random() })
  }

  /**
   * Devuelve y vacía la cola de notificaciones pendientes.
   * Llamado periódicamente por ScanProgressToast.vue.
   * @returns {Array<Object>} Notificaciones pendientes de mostrar.
   */
  const consumeNotifications = () => {
    const copy = [...pendingNotifications.value]
    pendingNotifications.value = []
    return copy
  }

  /**
   * Abre una conexión SSE con el backend para recibir el progreso del escaneo en tiempo real.
   * Actualiza scanProgressByAccount, y al recibir 'completed' o 'failed' cierra la conexión
   * y encola la notificación correspondiente.
   * @param {string} scanId - id del scan recién iniciado.
   * @param {string} accountId - id de la cuenta que se está escaneando.
   */
  const startSSE = (scanId, accountId) => {
    if (eventSources.has(accountId)) return

    startAccountScan(accountId, scanId)

    const token = localStorage.getItem('token')
    const url = buildApiSseUrl(`/cloud/scan_progress_sse/${scanId}`, { token })
    const es = new EventSource(url)
    eventSources.set(accountId, es)

    es.onmessage = (event) => {
      try {
        const data = JSON.parse(event.data)
        const progress = Number(data.progress ?? 0)
        setAccountScanProgress(accountId, progress)

        if (data.results) setScanData(accountId, scanId, data.results, data.created_at)

        if (data.status === 'completed') {
          es.close()
          eventSources.delete(accountId)
          completeAccountScan(accountId)
          pushNotification({ severity: 'success', summary: 'Escaneo Completo', detail: 'El escaneo de la cuenta ha finalizado correctamente', life: 4000 })
        }

        if (data.status === 'failed') {
          es.close()
          eventSources.delete(accountId)
          failAccountScan(accountId)
          pushNotification({ severity: 'error', summary: 'Escaneo fallido', detail: data.errors || 'Ocurrió un error durante el escaneo', life: 4000 })
        }
      } catch (e) {}
    }

    es.onerror = () => {
      es.close()
      eventSources.delete(accountId)
      failAccountScan(accountId)
      pushNotification({ severity: 'error', summary: 'Error de conexión', detail: 'Se perdió la conexión con el servidor durante el escaneo', life: 4000 })
    }
  }

  /**
   * Carga desde el backend el último scan de una cuenta y rellena el store.
   * Si el scan ya está cargado (scanIdByAccount tiene entrada), no hace la petición.
   * @param {Object|string} account - Cuenta (con .id o .account_id) o accountId directamente.
   * @returns {Promise<Object|null>} Datos del scan o null si no hay ninguno o hay error.
   */
  const loadScanDataForAccount = async (account) => {
    const accountId = account?.id || account?.account_id || account
    if (!accountId) { clearData(); return null }

    const token = localStorage.getItem('token')
    if (!token) { clearData(); return null }
    if (scanIdByAccount.value[accountId]) return
    try {
      const response = await fetch(buildApiUrl(`/cloud/get_scan_result/${accountId}`), {
        headers: { 'Authorization': `Bearer ${token}` }
      })
      const data = await response.json()
      if (!response.ok) throw new Error(data.detail || 'Error al cargar resultados')

      const scanId = data.scan_id || ''
      if (!scanId) { clearData(); return null }

      setScanData(accountId, scanId, data.results, data.created_at, data.context)
      console.log('Datos de escaneo cargados para la cuenta:', accountId, data)
      return data
    } catch (error) {
      console.error('Error cargando datos de escaneo:', error)
      clearData()
      return null
    }
  }

  /**
   * Comprueba si el último scan de una cuenta tiene menos de 24 horas de antigüedad.
   * Se usa en CloudAccountsView para advertir antes de lanzar un escaneo redundante.
   * @param {string} accountId
   * @returns {boolean} true si el scan existe y tiene menos de 24h.
   */
  const esScanReciente = (accountId) => {
    if (!scanCreatedAtByAccount.value[accountId]) return false
    const hours = (new Date() - new Date(scanCreatedAtByAccount.value[accountId])) / (1000 * 60 * 60)
    return hours < 24
  }

  /**
   * Inicia un escaneo AWS vía POST /cloud/start_scan y abre la conexión SSE de progreso.
   * Solo actúa si la cuenta es de tipo AWS.
   * @param {Object} account - Cuenta cloud a escanear (con .id y .provider).
   * @returns {Promise<void>}
   */
  const executeScan = async (account) => {
    if (account.provider !== 'AWS') return
    const token = localStorage.getItem('token')
    const response = await fetch(buildApiUrl('/cloud/start_scan'), {
      method: 'POST',
      headers: { 'Content-Type': 'application/json', Authorization: `Bearer ${token}` },
      body: JSON.stringify({ id: account.id })
    })
    const data = await response.json()
    if (!response.ok) throw new Error(data?.detail || 'No se pudo iniciar el escaneo')
    startSSE(data.scan_id, String(account.id))
  }

  return {
    scanResultIdByAccount,
    scanResultByAccount,
    scanCreatedAtByAccount,
    scanProgressByAccount,
    scanningAccounts,
    scanIdByAccount,
    pendingNotifications,
    resourceContextsByAccount,
    setScanData,
    startAccountScan,
    setAccountScanProgress,
    completeAccountScan,
    failAccountScan,
    clearData,
    startSSE,
    consumeNotifications,
    loadScanDataForAccount,
    setResourceContext,
    esScanReciente,
    executeScan
  }
})

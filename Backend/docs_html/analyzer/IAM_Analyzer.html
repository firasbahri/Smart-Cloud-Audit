 12    def analyze(self, users: list, groups: list, roles: list) -> list:
 13        """Pasa usuarios, grupos y roles por todas las comprobaciones IAM y junta los hallazgos.
 14
 15        Args:
 16            users (list): usuarios IAM ya convertidos al modelo de dominio (incluye el usuario root si se pudo leer).
 17            groups (list): grupos IAM del modelo de dominio.
 18            roles (list): roles IAM del modelo de dominio.
 19
 20        Returns:
 21            list[Vulnerability]: todos los hallazgos de check_user_permissions, check_mfa, check_inactive_users,
 22            check_group_permissions y check_role_permissions juntos.
 23        """
 24        vulnerabilities = []
 25        vulnerabilities.extend(self.check_user_permissions(users))
 26        vulnerabilities.extend(self.check_mfa(users))
 27        vulnerabilities.extend(self.check_inactive_users(users))
 28        vulnerabilities.extend(self.check_group_permissions(groups))
 29        vulnerabilities.extend(self.check_role_permissions(roles))
 30        return vulnerabilities

 32    def check_user_permissions(self, users):
 33        """Marca usuarios con AdministratorAccess o con una política inline que combine acción y recurso en wildcard (*).
 34
 35        Ambos casos se reportan como severidad Critical porque dan, en la práctica, control total de la cuenta.
 36
 37        Args:
 38            users (list): usuarios IAM a revisar, cada uno con managed_policies e inline_policies.
 39
 40        Returns:
 41            list[Vulnerability]: una entrada por cada usuario admin y otra por cada política inline con wildcard.
 42        """
 43        vulnerabilities = []
 44        for user in users:
 45            if self.isAdmin(user.managed_policies):
 46                logger.info(f"User {user.name} has AdministratorAccess managed policy")
 47                vulnerabilities.append(Vulnerability(
 48                    id=f"iam_user_{user.id}_managed_admin",
 49                    name="Usuario IAM con Política Administrada AdministratorAccess",
 50                    description=(
 51                        f"El usuario IAM '{user.name}' tiene la política administrada 'AdministratorAccess' adjunta, "
 52                        "lo que le otorga acceso completo a todos los servicios y recursos de AWS."
 53                    ),
 54                    severity="Critical",
 55                    resource_id=user.id,
 56                    resource_type="IAM User",
 57                ))
 58
 59            wildCardPolicies = self.hasWildcardPermissions(user.inline_policies)
 60            if wildCardPolicies:
 61                for pName in wildCardPolicies:
 62                    logger.info(f"User {user.name} has inline policy {pName} with wildcard permissions")
 63                    vulnerabilities.append(Vulnerability(
 64                        id=f"iam_user_{user.id}_inline_{pName}_wildcard",
 65                        name=f"Usuario IAM con Permisos Comodín en la Política Inline {pName}",
 66                        description=(
 67                            f"El usuario IAM '{user.name}' tiene una política inline '{pName}' que permite permisos comodín, "
 68                            "lo que puede generar privilegios excesivos y riesgos de seguridad."
 69                        ),
 70                        severity="Critical",
 71                        resource_id=user.id,
 72                        resource_type="IAM User",
 73                    ))
 74
 75        return vulnerabilities

149    def check_inactive_users(self, users):
150        """Busca usuarios que lleven más de 90 días sin usar su contraseña ni sus claves de acceso activas.
151
152        Una cuenta válida pero que nadie usa desde hace meses es un objetivo fácil si sus credenciales se filtran,
153        porque nadie va a notar actividad rara. Solo se marca si el usuario tiene al menos una clave de acceso activa
154        (si no tiene ninguna, no hay nada que comprometer).
155
156        Args:
157            users (list): usuarios IAM con password_last_used y access_keys.
158
159        Returns:
160            list[Vulnerability]: severidad Low, un hallazgo por usuario inactivo detectado.
161        """
162        vulnerabilities = []
163
164        for user in users:
165            active_user = False
166            days_password_active = False
167            days_access_key_active = False
168
169            if user.password_last_used:
170                days_password = (datetime.now(timezone.utc) - user.password_last_used).days
171                days_password_active = days_password > 90
172
173            if user.access_keys:
174                for access_key in user.access_keys:
175                    if access_key.get("Status") != "Active":
176                        continue
177
178                    create_date = access_key.get("CreateDate")
179                    if isinstance(create_date, str):
180                        try:
181                            create_date = datetime.fromisoformat(create_date.replace("Z", "+00:00"))
182                        except ValueError:
183                            create_date = None
184
185                    active_user = True
186                    if create_date:
187                        days_access_key = (datetime.now(timezone.utc) - create_date).days
188                        days_access_key_active = days_access_key > 90 or days_access_key_active
189
190            if (days_password_active or days_access_key_active) and active_user:
191                vulnerabilities.append(
192                    Vulnerability(
193                        id=f"iam_user_{user.id}_inactive",
194                        name="Usuario IAM Inactivo",
195                        description=(
196                            f"El usuario IAM '{user.name}' no ha utilizado su contraseña ni sus claves de acceso "
197                            "durante más de 90 días, lo que puede indicar que la cuenta está inactiva y representa un riesgo de seguridad."
198                        ),
199                        severity="Low",
200                        resource_id=user.id,
201                        resource_type="IAM User",
202                    )
203                )
204
205        logger.info("check_inactive_users processed")
206        return vulnerabilities

284    def isAdmin(self, policies: list) -> bool:
285        """True si alguna política administrada de la lista es exactamente 'AdministratorAccess'."""
286        logger.info(f"Checking policies: {policies}")
287        for policy in policies:
288            logger.info(f"Checking policy: {policy}")
289            if policy.get("policy_name") == "AdministratorAccess":
290                return True
291        return False

293    def hasWildcardPermissions(self, inline_policies: list) -> list:
294        """Devuelve los nombres de las políticas inline que permiten ('Allow') acción '*' sobre recurso '*' a la vez.
295
296        Args:
297            inline_policies (list): políticas inline ya normalizadas (con effect, actions y resources).
298
299        Returns:
300            list[str]: nombres de las políticas que combinan wildcard en acción y en recurso. Vacía si ninguna lo hace.
301        """
302        policiesWithWildcard = []
303        logger.info(f"Checking inline policies for wildcards: {inline_policies}")
304        for policy in inline_policies:
305            effect = policy.get("effect")
306            if effect != "Allow":
307                continue
308            action = policy.get("actions", [])
309            resources = policy.get("resources", [])
310            if "*" in action and "*" in resources:
311                policiesWithWildcard.append(policy.get("policy_name"))
312        return policiesWithWildcard

151        """Busca usuarios que lleven más de 90 días sin usar su contraseña ni sus claves de acceso activas.
152
153        Una cuenta válida pero que nadie usa desde hace meses es un objetivo fácil si sus credenciales se filtran,
154        porque nadie va a notar actividad rara. Solo se marca si el usuario tiene al menos una clave de acceso activa
155        (si no tiene ninguna, no hay nada que comprometer).
156
157        Args:
158            users (list): usuarios IAM con password_last_used y access_keys.
159
160        Returns:
161            list[Vulnerability]: severidad Low, un hallazgo por usuario inactivo detectado.
162        """
163        vulnerabilities = []
164
165        for user in users:
166            active_user = False
167            days_password_active = False
168            days_access_key_active = False
169
170            if user.password_last_used:
171                days_password = (datetime.now(timezone.utc) - user.password_last_used).days
172                days_password_active = days_password > 90
173
174            if user.access_keys:
175                for access_key in user.access_keys:
176                    if access_key.get("Status") != "Active":
177                        continue
178
179                    create_date = access_key.get("CreateDate")
180                    if isinstance(create_date, str):
181                        try:
182                            create_date = datetime.fromisoformat(create_date.replace("Z", "+00:00"))
183                        except ValueError:
184                            create_date = None
185
186                    active_user = True
187                    if create_date:
188                        days_access_key = (datetime.now(timezone.utc) - create_date).days
189                        days_access_key_active = days_access_key > 90 or days_access_key_active
190
191            if (days_password_active or days_access_key_active) and active_user:
192                vulnerabilities.append(
193                    Vulnerability(
194                        id=f"iam_user_{user.id}_inactive",
195                        name="Usuario IAM Inactivo",
196                        description=(

202                        resource_type="IAM User",
203                    )
